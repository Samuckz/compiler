// Teste 6 — Soma dos inteiros de 1 ate N
// Demonstra: if-else, repeat-until e operacoes aritmeticas

class Teste6 {
    int n, i, soma;
    {
        write("Digite um numero inteiro positivo: ");
        read(n);

        if (n > 0) {
            soma := 0;
            i := 1;
            repeat {
                soma := soma + i;
                i := i + 1;
            } until (i > n);
            write("A soma de 1 ate ");
            write(n);
            write(" e: ");
            write(soma);
        } else {
            write("Numero invalido. Digite um valor maior que zero.");
        };
    }
}
