class MinhaClasse
{
float a, b, c;
{
write("Digite um número");
read(a);
write("Digite outro número: ");
read(b);
write("Digite mais um número: ");
read(c);
maior := 0;
if (a > b) {
    if (a > c) {
        maior := a;
    } else {
        maior := c;
    };
} else {
    if (b > c) {
        maior := b;
    } else {
        maior := c;
    };
};
write("O maior número e: ");
write(maior);
}}
